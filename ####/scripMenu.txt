        document.querySelectorAll('.category-card, .footer a[data-category]').forEach(item => {

                const category = this.getAttribute('data-category');
                
                // Ocultar todas las categorías de productos
                document.querySelectorAll('section[id^="productos-"]').forEach(section => {
                    section.style.display = 'none';
                });
                
                // Mostrar la categoría seleccionada
                const targetSection = document.getElementById(`productos-${category}`);
                if (targetSection) {
                    targetSection.style.display = 'block';
                    
                    // Desplazar suavemente a la sección
                    targetSection.scrollIntoView({ behavior: 'smooth' });
                }
            });
        });

        // Funcionalidad para agregar productos al carrito
        document.querySelectorAll('.add-to-cart').forEach(button => {
            button.addEventListener('click', function() {
                const productCard = this.closest('.product-card');
                const productName = productCard.querySelector('.card-title').textContent;
                
                // Mostrar notificación
                const notification = document.querySelector('.notification');
                const notificationText = document.getElementById('notification-text');
                notificationText.textContent = `¡Agregaste ${productName} al carrito!`;
                notification.style.display = 'block';
                
                // Ocultar notificación después de 3 segundos
                setTimeout(() => {
                    notification.style.display = 'none';
                }, 3000);
            });
        });

        // Efecto de scroll suave para enlaces internos
        document.querySelectorAll('a[href^="#"]').forEach(anchor => {
            anchor.addEventListener('click', function (e) {
                e.preventDefault();
                
                const targetId = this.getAttribute('href');
                if (targetId === '#') return;
                
                const targetElement = document.querySelector(targetId);
                if (targetElement) {
                    targetElement.scrollIntoView({
                        behavior: 'smooth'
                    });
                }
            });
        });
    </script>

    <script>
    // Función para mostrar notificaciones
    function mostrarNotificacion(mensaje, tipo = 'success') {
        const notification = document.querySelector('.notification');
        const notificationText = document.getElementById('notification-text');
        notificationText.textContent = mensaje;
        notification.classList.remove('alert-success', 'alert-danger');
        notification.classList.add(`alert-${tipo}`);
        notification.style.display = 'block';
        setTimeout(() => {
            notification.style.display = 'none';
        }, 3000);
    }

    // Agregar al carrito con fetch
    document.querySelectorAll('.add-to-cart').forEach(button => {
        button.addEventListener('click', function(e) {
            e.preventDefault();  // Evitar comportamiento por defecto
            
            const productoId = this.dataset.productoId;
            const tamanoId = this.dataset.tamanoId;
            const url = this.dataset.url;
            
            if (!productoId || !tamanoId) {
                mostrarNotificacion('Error: Datos incompletos', 'danger');
                return;
            }

            fetch(url, {
                method: 'POST',
                headers: {
                    'X-CSRFToken': '{{ csrf_token }}',
                    'Content-Type': 'application/x-www-form-urlencoded',
                },
                body: new URLSearchParams({
                    'producto_id': productoId,
                    'tamano_id': tamanoId,
                    'cantidad': 1
                })
            })
            .then(response => response.json())
            .then(data => {
                if (data.success) {
                    mostrarNotificacion('Producto agregado al carrito');
                } else {
                    mostrarNotificacion('Error: ' + data.error, 'danger');
                }
            })
            .catch(error => {
                console.error('Error:', error);
                mostrarNotificacion('Error de conexión', 'danger');
            });
        });
    });

    // Scroll suave para enlaces internos
    document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function (e) {
            e.preventDefault();
            const targetId = this.getAttribute('href');
            if (targetId === '#') return;
            const targetElement = document.querySelector(targetId);
            if (targetElement) {
                targetElement.scrollIntoView({ behavior: 'smooth' });
            }
        });
    });
    </script>
